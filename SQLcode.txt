-- CREATE TABLES
CREATE TABLE foititis (
    id int auto_increment not null,
    email varchar(30) not null,
    password varchar(30) not null,
    name varchar(30) not null,
    lastname varchar(30) not null,
    address varchar(30) not null,
    phone varchar(10) not null,
    landline varchar(10) not null,
    primary key (id)
);


CREATE TABLE didaskwn (
    id int auto_increment not null,
    email varchar(30) not null,
    password varchar(30) not null,
    name varchar(30) not null,
    lastname varchar(30) not null,
    phone varchar(10) not null,
    primary key (id)
);

CREATE TABLE grammateia (
    id int auto_increment not null,
    email varchar(30) not null,
    password varchar(30) not null,
    name varchar(30) not null,
    lastname varchar(30) not null,
    primary key (id)
);

CREATE TABLE themata (
    id int auto_increment not null,
    didaskwn_id int not null,
    title varchar(300) not null,
    description varchar(300) not null,
    pdf varchar(300), 
    katastash enum('ΕΛΕΥΘΕΡΟ','ΔΕΣΜΕΥΜΕΝΟ'),
    foititis_id int,    
    primary key (id),
    foreign key (foititis_id) references foititis(id) on delete cascade on update cascade,
    foreign key (didaskwn_id) references didaskwn(id) on delete cascade on update cascade

);

CREATE TABLE diplwmatiki (
    id int auto_increment not null,
    title varchar(300) not null,
    foititis_id int not null,
    epivlepwn_id int not null,
    melos1_id int,
    melos2_id int,
    start_date date not null,
    katastash enum('ΥΠΟ ΑΝΑΘΕΣΗ','ΕΝΕΡΓΗ','ΥΠΟ ΕΞΕΤΑΣΗ','ΑΚΥΡΩΜΕΝΗ'),
    bathmos_enable boolean default false,
    bathmos_epivlepwn float check (bathmos_epivlepwn > 0 and bathmos_epivlepwn <= 10),
    bathmos_melos1 float check (bathmos_melos1 > 0 and bathmos_melos1 <= 10),
    bathmos_melos2 float check (bathmos_melos2 > 0 and bathmos_melos2 <= 10),
    proxeiro varchar(300),
    links varchar(300),
    arithmos_anathesis int,
    arithmos_akyrwshs int,
    etos_akyrwshs year,
    nhmerths_link varchar(255) not null,
    primary key (id),
    foreign key (foititis_id) references foititis(id) on delete cascade on update cascade,
    foreign key (epivlepwn_id) references didaskwn(id) on delete cascade on update cascade,
    foreign key (melos1_id) references didaskwn(id) on delete cascade on update cascade,
    foreign key (melos2_id) references didaskwn(id) on delete cascade on update cascade
);

CREATE TABLE diplwmatiki_FINISHED (
    diplwmatiki_id int auto_increment not null,
    title varchar(300) not null,
    foititis_id int not null,
    epivlepwn_id int not null,
    melos1_id int not null,
    melos2_id int not null,
    bathmos float not null check (bathmos > 0 and bathmos <= 10),
    nhmerths_link varchar(255) not null,
    praktiko varchar(255) not null,
    start_date date not null,
    end_date date not null,
    primary key (diplwmatiki_id)  
);

-- For each distinct epivlepwn_id we attach two notes referencing the smallest (earliest) diplwmatiki id they supervise.

SELECT t.epivlepwn_id, t.diplwmatiki_id, CONCAT('Σημείωση 1 από επιβλέποντα ', t.epivlepwn_id)
FROM (
    SELECT epivlepwn_id, MIN(id) AS diplwmatiki_id
    FROM diplwmatiki
    WHERE epivlepwn_id IS NOT NULL
    GROUP BY epivlepwn_id
) t;
